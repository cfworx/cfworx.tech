{{- /* Fenced code blocks render as console windows. The language picks the window: PowerShell gets the classic blue console, shells get a dark terminal, everything else a neutral console. Highlighting still comes from Chroma via the site config. */ -}} {{- $lang := .Type | default "text" -}} {{- $kind := "generic" -}} {{- $title := "console" -}} {{- if in (slice "powershell" "ps1" "pwsh") $lang -}} {{- $kind = "powershell" -}}{{- $title = "Windows PowerShell" -}} {{- else if in (slice "bash" "sh" "zsh" "shell" "console") $lang -}} {{- $kind = "bash" -}}{{- $title = $lang -}} {{- else if eq $lang "wow" -}} {{- $kind = "wow" -}}{{- $title = "World of Warcraft" -}} {{- else if ne $lang "text" -}} {{- $title = $lang -}} {{- end -}}
